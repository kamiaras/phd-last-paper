  \column{0.5}


  \block{Problem Setup}{
    Rounds $t=1,\ldots,T$.
    Choose allocation from budget before inflow or cost.

    \vspace{0.35em}
    \textbf{State \& action}
    \begin{itemize}
      \item Budget $b_t \geq 0$ at start of round $t$; $b_1=0$.
      \item Action $u_t \geq 0$ (outflow); feasibility: $0 \leq u_t \leq b_t$.
    \end{itemize}

    \vspace{0.25em}
    \textbf{Dynamics} (inflow $a_t \in [0,1]$ after $u_t$)
    \[
      b_{t+1} = b_t - u_t + a_t.
    \]

    \vspace{0.35em}
    \begin{center}
    \scalebox{1.35}{%
    \begin{tikzpicture}[
      font=\small,
      node/.style={draw=USCCardinal, rounded corners=4pt, line width=0.9pt,
                   minimum width=2.5cm, minimum height=1.8cm, align=center},
      flow/.style={-{Stealth[length=2.4mm]}, line width=0.9pt, USCDark},
    ]
      \node[node] (inflow) {$a_t$\\[-0.2em]{\scriptsize inflow}};
      \node[node, fill=USCCardinal!10, right=2.5cm of inflow] (state)
        {$b_t$\\[-0.2em]{\scriptsize budget}};
      \node[node, right=2.5cm of state] (action)
        {$u_t$\\[-0.2em]{\scriptsize action}};

      \draw[flow] (inflow) -- (state);
      \draw[flow] (state) -- (action);

      \node[below=0.8cm of state, font=\normalsize] {$b_{t+1} = b_t - u_t + a_t$};
    \end{tikzpicture}}
    \end{center}

    \vspace{0.25em}
    \textbf{Costs \& information}
    \begin{itemize}
      \item Cost $c_t(b_t,u_t)$; each $c_t$ convex in $(b,u)$.
      \item \textbf{Adversarial timing:} $u_t$ chosen \emph{before} $a_t$, $c_t$ observed.
    \end{itemize}

    \vspace{0.3em}
    \textbf{Objective:} min cumulative cost; feasible every round; compete with best fixed SDAC policy (below).
  }

  \block{Key Contributions}{
    \begin{itemize}
      \item \textbf{SDAC policy class} --- simplex-constrained disturbance-action policies that preserve feasibility by construction.
      \item \textbf{Online controller} over SDAC weights with sublinear horizon and logarithmic policy-dimension regret.
      \item \textbf{Adversarial setting} --- general convex state--action costs without distributional assumptions.
    \end{itemize}
  }

  

  \column{0.5}
  \block{Simplex Disturbance-Action Control (SDAC)}{
    \textbf{Idea:} constrain DAC's disturbance filter to the \emph{simplex} --- feasibility for free.

    \vspace{0.55em}
    \begin{center}
    \begin{tikzpicture}[
      font=\small,
      panel/.style={rounded corners=8pt, align=center, inner sep=14pt,
                    text width=0.34\linewidth},
      arr/.style={-{Stealth[length=7mm,width=6mm]}, line width=2.2pt, USCCardinal!85!black},
    ]
      \node[panel, draw=USCDark!45, line width=1.4pt, fill=USCDark!4] (dac) {%
        {\large\textbf{DAC}}\ {\footnotesize(standard)}\\[0.45em]
        $u_t=\kappa b_t+\sum_{i=1}^{H} (1-\kappa)^i m^{(i)} a_{t-i}$\\[0.3em]
        {\footnotesize filter $m\in\mathbb{R}^H$ unconstrained}\\[0.65em]
        \textcolor{vizred}{{\large$\boldsymbol{\times}$} may violate $0\le u_t\le b_t$}%
      };
      \node[panel, draw=USCCardinal, line width=2.4pt, fill=USCCardinal!8,
            right=5.5cm of dac] (sdac) {%
        {\large\textbf{SDAC}}\ {\footnotesize(this work)}\\[0.45em]
        $u_t=\kappa b_t+\sum_{i=1}^{H} (1-\kappa)^i w^{(i)} a_{t-i}$\\[0.3em]
        {\footnotesize weights $w\in\Delta_H$ (simplex)}\\[0.65em]
        \textcolor{feasgreen}{{\large$\checkmark$} feasible by construction}%
      };
      \draw[arr] (dac) -- node[midway, above, font=\footnotesize, text=USCDark!85]
        {constrain to $\Delta_H$} (sdac);
    \end{tikzpicture}
    \end{center}

    \vspace{0.45em}
    \begin{center}
    \begin{tikzpicture}[
      font=\small,
      callout/.style={rounded corners=8pt, align=left, inner sep=14pt,
                      text width=0.40\linewidth},
    ]
      \node[callout, draw=USCCardinal, line width=1.4pt, fill=USCCardinal!12] (c1) {%
        {\large\textbf{Feasible by construction}}\\[0.35em]
        Every fixed $w\in\Delta_H$ keeps $0\le u_t^{\pi(w)}\le b_t^{\pi(w)}$
        for all $t$.%
      };
      \node[callout, draw=USCGold!85!black, line width=1.4pt, fill=USCGold!30,
            right=2.5cm of c1] (c2) {%
        {\large\textbf{Convex \& online-learnable}}\\[0.35em]
        Fixed inflows $\Rightarrow b_t,u_t$ linear in $w \Rightarrow$ costs
        convex in $w \Rightarrow$ OCO on $\Delta_H$.%
      };
    \end{tikzpicture}
    \end{center}
  }
  

  \block{Online Algorithm}{
    \textbf{Inputs:} horizon $T$, step size $\eta>0$, gain $\kappa\in(0,1]$, memory $H$, initial weights $w_1\in\Delta_H$, $b_1=0$.

    \vspace{0.35em}
    \textbf{Each round $t=1,\ldots,T$} (choose action \emph{before} seeing $a_t,c_t$):

    \begin{enumerate}
      \item \textbf{Propose} an SDAC action from current weights:
        \[
          \widehat{u}_t \leftarrow \kappa b_t + \langle w_t,\widetilde{a}_t\rangle,
          \qquad
          \widetilde{a}_t^{(i)}=(1-\kappa)^i a_{t-i}.
        \]
      \item \textbf{Clip} to the available budget (feasibility):
        \[
          u_t \leftarrow \min\{\widehat{u}_t,\, b_t\}.
        \]
      \item \textbf{Update state} after inflow $a_t$ arrives:
        \[
          b_{t+1} \leftarrow b_t - u_t + a_t.
        \]
      \item \textbf{Learn} from the revealed cost $c_t(b_t,u_t)$: build a convex surrogate
        $\ell_t(w)=c_t\bigl(b_t^{\pi(w)},u_t^{\pi(w)}\bigr)$ and take one exponentiated-gradient step on the simplex,
        \[
          w_{t+1}^{(i)} \propto w_t^{(i)}\exp\!\bigl(-\eta\, g_t^{(i)}\bigr),
          \qquad g_t\in\partial\ell_t(w_t),
        \]
        then renormalize so $w_{t+1}\in\Delta_H$.
    \end{enumerate}
  }

  \block{Regret Guarantee}{
    \textbf{Assumption.} Each $c_t$ is convex and Lipschitz.

    \vspace{0.3em}
    \textbf{Benchmark.} Compare cumulative cost to the best \emph{fixed} SDAC policy in hindsight:
    \begin{center}
    \scalebox{1}{%
      \mbox{\boldmath$\displaystyle
        \mathrm{Regret}(T)
        \triangleq
        \sum_{t=1}^T c_t(b_t,u_t)
        -
        \min_{w\in\Delta_H}\sum_{t=1}^T c_t\bigl(b_t^{\pi(w)},u_t^{\pi(w)}\bigr)
      $}%
    }
    \end{center}

    \vspace{0.3em}
    \textbf{Main bound.} The clipped exponentiated-gradient algorithm satisfies,
    \begin{center}
    \scalebox{1.5}{%
      \mbox{\boldmath$\displaystyle
        \mathrm{Regret}(T)
        =
        O\!\left(\tfrac{\sqrt{T\log H}}{\kappa^2}\right)
      $}%
    }
    \end{center}
  }
