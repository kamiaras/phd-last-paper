  \block{Problem Setup}{
    \textbf{Dynamics}
    \begin{center}
    \scalebox{1.35}{%
    \begin{tikzpicture}[
      font=\small,
      node/.style={draw=USCCardinal, rounded corners=4pt, line width=0.9pt,
                   minimum width=2.5cm, minimum height=1.8cm, align=center},
      flow/.style={-{Stealth[length=2.4mm]}, line width=0.9pt, USCDark},
    ]
      \node[node] (inflow) {$a_t$\\[-0.2em]{\scriptsize inflow}};
      \node[node, fill=USCCardinal!10, right=2.5cm of inflow] (state)
        {$b_t$\\[-0.2em]{\scriptsize budget}};
      \node[node, right=2.5cm of state] (action)
        {$u_t$\\[-0.2em]{\scriptsize action}};

      \draw[flow] (inflow) -- (state);
      \draw[flow] (state) -- (action);

      \node[below=0.8cm of state, font=\normalsize] {$b_{t+1} = b_t - u_t + a_t$};
    \end{tikzpicture}}
    \end{center}

    \vspace{0.35em}
    \textbf{Constraints}
    \begin{itemize}
      \item Budget $b_t \geq 0$.
      \item Feasibility: $0 \leq u_t \leq b_t$.
      \item Inflow $a_t \in [0,1]$.
    \end{itemize}

    \vspace{0.35em}
    \textbf{Game} (rounds $t=1,\ldots,T$)
    \begin{enumerate}
      \item Controller chooses $u_t$.
      \item Adversary chooses $a_t$ and $c_t$; player suffers cost $c_t(b_t,u_t)$.
      \item State updates via dynamics; repeat.
    \end{enumerate}
  }

  


  

  \block{Simplex Disturbance-Action Control (SDAC)}{
    \begin{center}
    {\large\textbf{DAC.}} Disturbance-Action Control
    \[
      u_t = \kappa b_t + \sum_{i=1}^{H} m^{(i)} a_{t-i},
      \qquad m \in \mathbb{R}^H.
    \]
    \end{center}

    \vspace{0.25em}
    \begin{center}
    \begin{tikzpicture}
      \node[font=\large, text=vizred] (warn) {%
        Does not guarantee $0 \le u_t \le b_t$%
      };
      \draw[-{Stealth[length=5mm,width=4mm]}, line width=2pt, USCCardinal!80!black]
        (warn.south) -- ++(0,-1.1);
    \end{tikzpicture}
    \end{center}

    \vspace{0.15em}
    \begin{center}
    \begin{tikzpicture}
      \node[
        rounded corners=12pt,
        draw=USCCardinal,
        line width=3pt,
        fill=USCGold!32,
        inner sep=16pt,
        text width=0.94\linewidth,
        align=center,
      ] {%
        {\LARGE\textbf{\textcolor{USCCardinal}{SDAC}}}\\[-0.05em]
        {\large Bound filter on the simplex, \mbox{$w\in\Delta_H$}}\\[0.35em]
        \scalebox{1.22}{$\displaystyle
          u_t = \kappa b_t + \sum_{i=1}^{H} (1-\kappa)^i w^{(i)} a_{t-i}.
        $}%
      };
    \end{tikzpicture}
    \end{center}
  }
  
  \column{0.5}
  
  \block{Motivational Applications}{
    \applicationflow{Energy harvesting}{arrivals}{battery}{discharge}
    \applicationflow{Queueing}{arrivals}{backlog}{service}
    \applicationflow{Inventory control}{restock}{stock}{shipments}
    \applicationflow{Bank accounts}{deposits}{balance}{withdrawals}
    \applicationflow{Water reservoirs}{inflow}{storage}{release}
  }

  \block{Online Algorithm}{
    \textbf{Inputs:} \small{horizon $T$, step size $\eta>0$, gain $\kappa\in(0,1]$, memory $H$, initial weights $w_1\in\Delta_H$, $b_1=0$}.
    \normalsize
    \vspace{0.35em}

    \begin{enumerate}
      \item \textbf{Choose} an SDAC action from current weights:
        \[
          \widehat{u}_t \leftarrow \kappa b_t + \langle w_t,\widetilde{a}_t\rangle,
          \qquad
          \widetilde{a}_t^{(i)}=(1-\kappa)^i a_{t-i}.
        \]
      \item \textbf{Clip} to the available budget (feasibility):
        \[
          u_t \leftarrow \min\{\widehat{u}_t,\, b_t\}.
        \]
      \item \textbf{Update state} after inflow $a_t$ arrives:
        \[
          b_{t+1} \leftarrow b_t - u_t + a_t.
        \]
      \item \textbf{Learn} from the revealed cost $c_t(b_t,u_t)$: build a convex surrogate
        $\ell_t(w)=c_t\bigl(b_t^{\pi(w)},u_t^{\pi(w)}\bigr)$ and take one exponentiated-gradient step on the simplex,
        \[
          w_{t+1}^{(i)} \propto w_t^{(i)}\exp\!\bigl(-\eta\, g_t^{(i)}\bigr),
          \qquad g_t\in\partial\ell_t(w_t).
        \]
    \end{enumerate}
  }

  \block{Regret Guarantee}{
    \textbf{Assumption.} Each $c_t$ is convex and Lipschitz.

    \vspace{0.3em}
    \textbf{Regret bound.} Against the best \emph{fixed} SDAC policy in hindsight, the clipped exponentiated-gradient algorithm satisfies
    \begin{center}
    \scalebox{0.8}{%
      \mbox{\boldmath$\displaystyle
        \mathrm{Regret}(T)
        \triangleq
        \sum_{t=1}^T c_t(b_t,u_t)
        -
        \min_{w\in\Delta_H}\sum_{t=1}^T c_t\bigl(b_t^{\pi(w)},u_t^{\pi(w)}\bigr)
        =
        O\!\left(\tfrac{\sqrt{T\log H}}{\kappa^2}\right)
      $}%
    }
    \end{center}
  }
